// src/app/api/analysis-history/route.js
import { NextResponse } from 'next/server';
import fs from 'fs';
import path from 'path';

// In a real application, you would use a database
// For this example, we'll use a simple JSON file
const DATA_FILE = path.join(process.cwd(), 'data', 'analysis-history.json');

// Ensure data directory exists
const ensureDataDir = () => {
  const dataDir = path.join(process.cwd(), 'data');
  if (!fs.existsSync(dataDir)) {
    fs.mkdirSync(dataDir);
  }
  
  if (!fs.existsSync(DATA_FILE)) {
    fs.writeFileSync(DATA_FILE, JSON.stringify([]));
  }
};

// Get all analysis history
export async function GET() {
  try {
    ensureDataDir();
    
    const data = fs.readFileSync(DATA_FILE, 'utf8');
    const history = JSON.parse(data);
    
    // Sort by date, newest first
    history.sort((a, b) => new Date(b.date) - new Date(a.date));
    
    return NextResponse.json(history);
  } catch (error) {
    console.error('Error fetching analysis history:', error);
    return NextResponse.json({ error: 'Failed to fetch analysis history' }, { status: 500 });
  }
}

// src/app/api/save-analysis/route.js
export async function POST(req) {
  try {
    const analysisData = await req.json();
    
    if (!analysisData.url || !analysisData.title) {
      return NextResponse.json({ error: 'URL and title are required' }, { status: 400 });
    }
    
    ensureDataDir();
    
    // Read existing data
    const data = fs.readFileSync(DATA_FILE, 'utf8');
    const history = JSON.parse(data);
    
    // Check if an analysis with this URL already exists
    const existingIndex = history.findIndex(item => item.url === analysisData.url);
    
    if (existingIndex !== -1) {
      // Update existing entry
      const existingEntry = history[existingIndex];
      
      // Preserve existing QA history if available and merge with new QA history
      const existingQaHistory = existingEntry.qaHistory || [];
      const newQaHistory = analysisData.qaHistory || [];
      
      // Merge QA histories, avoiding duplicates
      const mergedQaHistory = [...existingQaHistory];
      
      // Only add questions that don't already exist
      for (const qa of newQaHistory) {
        const exists = mergedQaHistory.some(
          existing => existing.question === qa.question && existing.answer === qa.answer
        );
        
        if (!exists) {
          mergedQaHistory.push(qa);
        }
      }
      
      // Update the existing entry
      history[existingIndex] = {
        ...existingEntry,
        title: analysisData.title,
        thumbnail: analysisData.thumbnail || existingEntry.thumbnail,
        summary: analysisData.summary,
        date: new Date().toISOString(), // Update the date to show it was recently analyzed
        qaHistory: mergedQaHistory
      };
      
      // Save updated data
      fs.writeFileSync(DATA_FILE, JSON.stringify(history, null, 2));
      
      return NextResponse.json({ success: true, id: existingEntry.id, updated: true });
    } else {
      // Add new analysis with a unique ID
      const newAnalysis = {
        id: Date.now().toString(),
        ...analysisData,
        date: analysisData.date || new Date().toISOString(),
        thumbnail: analysisData.thumbnail || null,
        qaHistory: analysisData.qaHistory || []
      };
      
      history.push(newAnalysis);
      
      // Save updated data
      fs.writeFileSync(DATA_FILE, JSON.stringify(history, null, 2));
      
      return NextResponse.json({ success: true, id: newAnalysis.id, updated: false });
    }
  } catch (error) {
    console.error('Error saving analysis:', error);
    return NextResponse.json({ error: 'Failed to save analysis' }, { status: 500 });
  }
}
